cmake_minimum_required(VERSION 3.15)
project(MemoryReader CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(memreader main.cpp)
if(MSVC)
  target_compile_options(memreader PRIVATE /W4 /utf-8)
  target_compile_definitions(memreader PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
